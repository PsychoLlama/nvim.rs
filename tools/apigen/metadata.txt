# The parts of the api-info metadata blob that no Rust source declares.
#
# `nvim --api-info` and `nvim_get_api_info()` return one msgpack dict with six
# sections. tools/apigen builds the `functions` section from
# tools/apigen/functions.txt plus the API signatures, and `ui_options` from
# `ui_ext_names` in crates/nvim/src/nvim/main.rs. The rest have no source of
# truth in the tree, so they are spelled out here — the point is that one
# audited encoder produces the blob and the committed bytes are checked, not
# that every value gains a second home.
#
# Lines are `<section> <name> [key=value ...]`.

# The `version` dict, minus major/minor/patch, which are read from
# NVIM_VERSION_MAJOR/MINOR/PATCH in crates/nvim/src/nvim/version.rs.
#
#   api_level       the API level this build implements
#   api_compatible  the oldest API level it stays compatible with
#   api_prerelease  whether api_level is still being changed
#   prerelease      whether this is a development build
#   build           the build's version string, upstream's NVIM_VERSION_BUILD

version api_level=14 api_compatible=0 api_prerelease=false prerelease=false build=v0.12.4

# The error kinds an RPC error response can carry, by their wire ids.

error_type Exception id=0
error_type Validation id=1

# The handle types the API takes and returns. `prefix` is also what makes a
# method a "method" of its type: `method` is true in the metadata for exactly
# the functions whose name starts with one of these.

type Buffer id=0 prefix=nvim_buf_
type Window id=1 prefix=nvim_win_
type Tabpage id=2 prefix=nvim_tabpage_
